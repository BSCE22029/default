Scheduler Core — Best-Fit Bin Packing
// scheduler.js — Muhammad Moiz Ahmad
// Picks the node with the LEAST leftover capacity that can still fit the
// pod (best-fit), which packs bins tighter than first-fit / random and
// leaves larger contiguous free space on other nodes for big pods later.

function scheduleBestFit(pod, nodes) {
  let best = null;
  let bestLeftover = Infinity;

  for (const node of nodes) {
    if (node.status !== 'ready') continue;

    const freeCpu = node.cpuCap - node.cpuUsed;
    const freeMem = node.memCap - node.memUsed;
    const fits = freeCpu >= pod.cpu && freeMem >= pod.mem;
    if (!fits) continue;

    // leftover capacity if this pod lands here — smaller is "tighter fit"
    const leftover = (freeCpu - pod.cpu) + (freeMem - pod.mem);
    if (leftover < bestLeftover) {
      bestLeftover = leftover;
      best = node;
    }
  }
  return best; // null => pod goes to the pending queue
}

function onNodeFailure(deadNode, nodes, pendingPods) {
  // evict every pod that was on the dead node, then re-run the scheduler
  const orphans = deadNode.pods.splice(0);
  deadNode.status = 'dead';
  for (const pod of orphans) {
    const target = scheduleBestFit(pod, nodes);
    target ? placePod(pod, target) : pendingPods.push(pod);
  }
}

function maybeAutoscale(nodes, threshold = 0.75) {
  const live = nodes.filter(n => n.status === 'ready');
  const avgCpuUtil = live.reduce((s,n) => s + n.cpuUsed/n.cpuCap, 0) / live.length;
  if (avgCpuUtil >= threshold) {
    nodes.push(makeNode()); // horizontal scale-out
    drainPendingQueue(nodes);
  }
}
