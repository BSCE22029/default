Verilog Implementation
// Processing Element — core of systolic array
// Muhammad Moiz Ahmad — FPGA Neural Network Accelerator

module pe #(parameter DATA_W = 8, ACC_W = 32)(
    input                clk, rst, valid_in,
    input  [DATA_W-1:0]  data_in,    // activation (flows right)
    input  [DATA_W-1:0]  weight_in,  // weight (flows down)
    output reg [DATA_W-1:0]  data_out, weight_out,
    output reg [ACC_W-1:0]   acc_out,
    output reg               valid_out
);
reg [ACC_W-1:0] acc;
always @(posedge clk) begin
    if (rst) acc <= 0;
    else if (valid_in)
        acc <= acc + $signed(data_in) * $signed(weight_in);
    data_out   <= data_in;
    weight_out <= weight_in;
    valid_out  <= valid_in;
    acc_out    <= acc;
end
endmodule

// ReLU Activation Unit
module relu_unit #(parameter W = 32)(
    input  [W-1:0] acc_in,
    output [W-1:0] relu_out
);
assign relu_out = acc_in[W-1] ? 0 : acc_in;  // zero if negative
endmodule
