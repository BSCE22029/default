# Multi-agent debate graph — Muhammad Moiz Ahmad
# Optimist and Skeptic alternate for N rounds against a shared DebateState,
# each reading the other's last argument before responding; Moderator judges.

class DebateState(TypedDict):
    topic: str
    round: int
    max_rounds: int
    transcript: list[dict]   # {speaker, text}
    consensus: dict | None

def optimist_node(state: DebateState) -> DebateState:
    # Reads the Skeptic's last rebuttal (if any) and argues the upside case
    last_rebuttal = last_turn(state["transcript"], speaker="skeptic")
    argument = llm.argue(
        persona="optimist", topic=state["topic"],
        respond_to=last_rebuttal, stance="favor upside, momentum, opportunity cost"
    )
    state["transcript"].append({"speaker": "optimist", "text": argument})
    return state

def skeptic_node(state: DebateState) -> DebateState:
    # Reads the Optimist's latest point and counters with risk/cost framing
    last_claim = last_turn(state["transcript"], speaker="optimist")
    rebuttal = llm.argue(
        persona="skeptic", topic=state["topic"],
        respond_to=last_claim, stance="surface risk, hidden cost, downside scenarios"
    )
    state["transcript"].append({"speaker": "skeptic", "text": rebuttal})
    state["round"] += 1
    return state

def should_continue(state: DebateState) -> str:
    if state["round"] >= state["max_rounds"]:
        return "moderator"
    return "optimist"  # loop for another round

def moderator_node(state: DebateState) -> DebateState:
    # Weighs the full transcript from both sides, not just a vote count
    state["consensus"] = llm.synthesize(
        topic=state["topic"], transcript=state["transcript"],
        instructions="weigh both sides, state conditions under which each wins, give one recommendation"
    )
    return state

graph = StateGraph(DebateState)
graph.add_node("optimist", optimist_node)
graph.add_node("skeptic", skeptic_node)
graph.add_node("moderator", moderator_node)
graph.add_conditional_edges("skeptic", should_continue)
app = graph.compile()
