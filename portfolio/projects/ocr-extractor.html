// ocr.js — Muhammad Moiz Ahmad
// Runs entirely client-side: Tesseract.js compiles the Tesseract OCR
// engine to WebAssembly and ships the "eng" traineddata over CDN.

async function extractText(imageSourceOrFile) {
  const startedAt = Date.now();

  const { data } = await Tesseract.recognize(
    imageSourceOrFile,
    'eng',
    {
      logger: (m) => {
        // m looks like: { status: 'recognizing text', progress: 0.42 }
        updateProgressBar(m.status, m.progress);
        appendLogLine(`[${m.status}] ${Math.round((m.progress||0)*100)}%`);
      },
    }
  );

  const elapsedMs = Date.now() - startedAt;
  return {
    text: data.text,
    confidence: data.confidence,          // 0-100, from Tesseract
    wordCount: data.text.trim().split(/\s+/).filter(Boolean).length,
    elapsedMs,
  };
}
