// Real parse + execute, not a lookup table of canned results.

function execute(sql, table) {
  const { cols, where, groupBy, orderBy, limit } = parse(sql);
  let rows = table.filter(row => !where || evalWhere(where, row));

  if (groupBy) {
    rows = aggregate(rows, groupBy, cols);   // COUNT / SUM / AVG / MIN / MAX
  } else {
    rows = rows.map(row => project(row, cols));
  }
  if (orderBy) rows.sort(comparator(orderBy));
  if (limit)   rows = rows.slice(0, limit);
  return rows;
}
