Transform & Validation Rules (JavaScript)
// etl.js — Muhammad Moiz Ahmad
// Real rules applied to every row — no mock output.

const RULES = {
  email: v => /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(v),
  qty:   v => Number.isFinite(Number(v)) && Number(v) >= 0,
  date:  v => !isNaN(Date.parse(v)),
};

function rowHash(row) {
  // Normalize before hashing so "  Ali@X.com " and "ali@x.com" collide
  return Object.values(row)
    .map(v => String(v).trim().toLowerCase())
    .join('|');
}

function normalizeDate(v) {
  // Rearrange components directly — going through Date/toISOString()
  // shifts the calendar day whenever the local timezone is behind UTC.
  const mdy = v.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})$/);
  if (mdy) {
    const [, m, d, y] = mdy;
    return `${y}-${m.padStart(2,'0')}-${d.padStart(2,'0')}`;
  }
  return v; // already ISO, or left as-is for the validator to flag
}
