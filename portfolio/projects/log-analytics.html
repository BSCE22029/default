Log Line Parser (JavaScript)
// log_parser.js — Muhammad Moiz Ahmad
// Nginx/Apache "combined" log format, parsed with one real regex.

const LOG_RE = /^(\S+) \S+ \S+ \[([^\]]+)\] "(\S+) (\S+) \S+" (\d{3}) (\d+) (\d+)$/;

function parseLine(line) {
  const m = line.match(LOG_RE);
  if (!m) return null;
  const [, ip, ts, method, path, status, bytes, ms] = m;
  return { ip, ts, method, path, status: Number(status), bytes: Number(bytes), ms: Number(ms) };
}
