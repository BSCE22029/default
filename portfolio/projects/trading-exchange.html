Matching Loop + Double-Entry Settlement (JavaScript)
// Price-time-priority match: walk the resting contra-side book,
// filling at the RESTING (maker) order's price, until the
// incoming order is filled or the book stops crossing.
while (remaining > 0 && contraSide.length > 0) {
    const best = contraSide[0];
    const crosses = type === 'MARKET' ||
        (side === 'BUY'  && price >= best.price) ||
        (side === 'SELL' && price <= best.price);
    if (!crosses) break;

    const fillQty  = Math.min(remaining, best.qty);
    const notional = fillQty * best.price;

    // ── settle: 4 balanced entries per trade, debits === credits ──
    postLedger(tradeId, [
        { account: buyer,  line: symbol, debit: fillQty,  credit: 0 },
        { account: buyer,  line: 'CASH',  debit: 0,      credit: notional },
        { account: seller, line: 'CASH',  debit: notional, credit: 0 },
        { account: seller, line: symbol, debit: 0,      credit: fillQty },
    ]);

    remaining -= fillQty;
    best.qty  -= fillQty;
    if (best.qty <= 0) contraSide.shift();  // resting order fully consumed
}
if (remaining > 0 && type === 'LIMIT') insertSorted(book, order); // rest the remainder
