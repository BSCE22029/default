// PWM Generator — Verilog HDL
// Variable duty cycle and frequency
// Author: Muhammad Moiz Ahmad

module pwm_gen #(
    parameter CLK_FREQ = 50_000_000,  // 50 MHz
    parameter PWM_FREQ = 1000          // 1 kHz
)(
    input        clk, rst,
    input  [7:0] duty,   // 0–255 (0–100%)
    output reg   pwm_out
);

localparam PERIOD = CLK_FREQ / PWM_FREQ;   // 50000 cycles
reg [15:0] counter;
wire [15:0] threshold = (PERIOD * duty) >> 8;

always @(posedge clk) begin
    if (rst) begin
        counter <= 0; pwm_out <= 0;
    end else begin
        counter <= (counter == PERIOD-1) ? 0 : counter+1;
        pwm_out <= (counter < threshold);
    end
end
endmodule
