// edge_anomaly.cpp — Autoencoder on ESP32 + ADXL345
// Muhammad Moiz Ahmad
#include <TensorFlowLite_ESP32.h>
#include <Adafruit_ADXL345_U.h>
#include "autoencoder_model.h"

constexpr float THRESHOLD  = 0.45f;
constexpr int   WINDOW_LEN = 128;

float computeReconError(float* input, float* output, int len) {
    float mse = 0;
    for (int i = 0; i < len; i++)
        mse += (input[i] - output[i]) * (input[i] - output[i]);
    return mse / len;
}

void loop() {
    // Collect 128 accelerometer samples
    float window[WINDOW_LEN * 3];
    collectIMUWindow(window, WINDOW_LEN);
    normalizeWindow(window, WINDOW_LEN * 3);

    // Run autoencoder: encode → decode
    setInput(window, WINDOW_LEN * 3);
    interpreter->Invoke();
    float* recon = getOutput();

    float err = computeReconError(window, recon, WINDOW_LEN * 3);

    if (err > THRESHOLD) {
        Serial.printf("[ANOMALY] Error=%.3f — fault detected!\n", err);
        triggerAlert();   // buzzer + LED
    } else {
        Serial.printf("[NORMAL]  Error=%.3f\n", err);
    }
}
