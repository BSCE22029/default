Self-Correcting Code Agent

Unlike a multi-agent orchestrator that routes a task between planner/tool/critic roles, this is a single coding agent in a tight generate-execute-verify loop. It writes a candidate function, actually runs it in a sandboxed new Function(...) against real test cases in your browser, reads the real pass/fail output, and — if it fails — rewrites the function and tries again. Nothing here is a scripted pass/fail string; every result below comes from code that really executed.

// This is the real function this page calls to run each candidate attempt.
// It is not a mock — candidateSource is genuine JS text, compiled with
// new Function() and invoked once per test case inside a try/catch.

function runCandidate(candidateSource, testCases) {
  let compiled;
  try {
    // Compile the agent's candidate source into a callable function.
    // "return (" + source + ")" turns a function expression/declaration
    // string into a value we can invoke directly — no eval() needed.
    compiled = new Function('"use strict"; return (' + candidateSource + ');')();
  } catch (compileErr) {
    // A syntax error in the candidate is itself a real failure signal
    return testCases.map(tc => ({ ...tc, actual: undefined,
      pass: false, error: 'CompileError: ' + compileErr.message }));
  }

  return testCases.map(tc => {
    try {
      const actual = compiled(...tc.args);
      const pass = JSON.stringify(actual) === JSON.stringify(tc.expected);
      return { ...tc, actual, pass, error: null };
    } catch (runErr) {
      // Thrown errors (e.g. undefined.length) are caught here so a
      // broken candidate can never crash the page — it just fails the test.
      return { ...tc, actual: undefined, pass: false, error: runErr.message };
    }
  });
}

// Agent loop: generate -> execute -> verify -> retry on real failure
async function agentLoop(attempts, testCases) {
  for (let i = 0; i < attempts.length; i++) {
    const results = runCandidate(attempts[i].source, testCases);
    const allPass = results.every(r => r.pass);
    if (allPass) return { attempt: i, results, done: true };
    // real failure -> agent writes its next hand-authored fix and retries
  }
  return { attempt: attempts.length - 1, results: null, done: false };
}
