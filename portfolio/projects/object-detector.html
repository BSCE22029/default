// object-detector.html — Muhammad Moiz Ahmad
// Loads COCO-SSD once, then runs detect() on either a <video> stream
// or a static <img>, and draws boxes on a <canvas> overlay.

let model = null;

async function loadModel() {
  model = await cocoSsd.load();  // downloads + initializes the SSD weights
}

async function detectOnVideo(video, canvas) {
  const ctx = canvas.getContext('2d');
  const t0 = performance.now();
  const predictions = await model.detect(video);
  const ms = performance.now() - t0;

  ctx.clearRect(0, 0, canvas.width, canvas.height);
  for (const p of predictions) {
    const [x, y, w, h] = p.bbox;           // {bbox:[x,y,w,h], class, score}
    ctx.strokeStyle = colorFor(p.class);
    ctx.lineWidth = 2;
    ctx.strokeRect(x, y, w, h);
    ctx.fillStyle = colorFor(p.class);
    ctx.fillText(`${p.class} ${(p.score*100).toFixed(0)}%`, x + 4, y - 6);
  }
  return { predictions, ms };
}

// Live loop for webcam mode — re-runs detection every animation frame
function renderLoop(video, canvas) {
  async function tick() {
    if (!streaming) return;
    const { predictions, ms } = await detectOnVideo(video, canvas);
    updateMetrics(predictions.length, ms);
    requestAnimationFrame(tick);
  }
  tick();
}
