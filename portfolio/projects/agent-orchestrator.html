# Agent graph definition — Muhammad Moiz Ahmad
# Planner -> Tool-Executor -> Critic, with conditional retry loop

class AgentState(TypedDict):
    task: str
    plan: list[str]
    tool_results: list[dict]
    critique: str | None
    retries: int

def planner_node(state: AgentState) -> AgentState:
    # LLM call: decompose task into ordered tool-calling steps
    state["plan"] = llm.plan(state["task"])
    return state

def tool_executor_node(state: AgentState) -> AgentState:
    for step in state["plan"]:
        result = tools[step.tool_name](**step.args)
        state["tool_results"].append(result)
    return state

def critic_node(state: AgentState) -> AgentState:
    # LLM call: does the result actually satisfy the task?
    state["critique"] = llm.critique(state["task"], state["tool_results"])
    return state

def should_retry(state: AgentState) -> str:
    if state["critique"].passed or state["retries"] >= 2:
        return "final_answer"
    state["retries"] += 1
    return "planner"  # loop back with critic feedback

graph = StateGraph(AgentState)
graph.add_node("planner", planner_node)
graph.add_node("tool_executor", tool_executor_node)
graph.add_node("critic", critic_node)
graph.add_conditional_edges("critic", should_retry)
app = graph.compile()
