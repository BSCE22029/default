// keyword_spotter.c — STM32F4 + I2S microphone
// Muhammad Moiz Ahmad — Wake word detection
#include "arm_math.h"
#include "mfcc.h"
#include "ds_cnn_model.h"

#define SAMPLE_RATE   16000
#define FRAME_LEN     400   // 25ms
#define HOP_LEN       160   // 10ms
#define NUM_MFCC      40
#define NUM_FRAMES    49

static float mfcc_buf[NUM_MFCC * NUM_FRAMES];
static float audio_buf[SAMPLE_RATE];  // 1s rolling window

void process_audio(int16_t* pcm, int len) {
    // 1. Pre-emphasis filter
    for (int i = len-1; i > 0; i--)
        pcm[i] -= (int16_t)(0.97f * pcm[i-1]);

    // 2. Compute MFCC (uses ARM CMSIS-DSP FFT)
    mfcc_compute(pcm, len, mfcc_buf, NUM_MFCC, NUM_FRAMES);

    // 3. Run DS-CNN inference (INT8 quantized)
    float probs[8];
    ds_cnn_infer(mfcc_buf, probs);

    int best = argmax(probs, 8);
    if (best == WAKE_WORD_IDX && probs[best] > 0.85f)
        trigger_wake_event();  // activate assistant
}
