Real-Time Collaborative Editor

A genuine Replicated Growable Array (RGA) CRDT — the same class of algorithm behind Google Docs / Figma-style multiplayer editing. Each client generates real operations with unique (site, counter) identifiers and tombstoned deletes; a deterministic merge order guarantees every replica converges to the identical text no matter what order operations arrive in, even after a simulated network partition.

// rga.js — Replicated Growable Array CRDT — Muhammad Moiz Ahmad
// Each character is a node keyed by a globally unique id = (siteId, counter).
// Nodes are never removed on delete — they're tombstoned — so every replica
// keeps the same causal history and can always re-derive the same order.

class RGA {
  constructor(siteId) {
    this.siteId = siteId;
    this.counter = 0;
    // linked list of nodes: { id:{site,counter}, char, tombstone, leftOrigin }
    this.nodes = [];   // kept in RGA-order at all times
  }

  nextId() {
    this.counter += 1;
    return { site: this.siteId, counter: this.counter };
  }

  // Deterministic total order used to place concurrent inserts at the same
  // origin: higher counter wins, ties broken by siteId (lexicographic).
  compareId(a, b) {
    if (a.counter !== b.counter) return b.counter - a.counter;
    return a.site < b.site ? -1 : a.site > b.site ? 1 : 0;
  }

  // Insert a character locally after visible-position `afterIdx`.
  // Produces an op that can be broadcast and replayed on any replica.
  localInsert(afterIdx, char) {
    const leftOrigin = afterIdx >= 0 ? this.visibleNodeAt(afterIdx).id : null;
    const id = this.nextId();
    const node = { id, char, tombstone: false, leftOrigin };
    this.integrate(node);
    return { type: 'insert', id, char, leftOrigin };
  }

  localDelete(idx) {
    const node = this.visibleNodeAt(idx);
    node.tombstone = true;
    return { type: 'delete', id: node.id };
  }

  // Apply a remote op — insert or delete — idempotently and order-independently.
  applyRemote(op) {
    if (op.type === 'insert') {
      if (this.nodes.some(n => n.id.site===op.id.site && n.id.counter===op.id.counter)) return; // dedupe
      this.integrate({ id: op.id, char: op.char, tombstone: false, leftOrigin: op.leftOrigin });
    } else {
      const n = this.nodes.find(n => n.id.site===op.id.site && n.id.counter===op.id.counter);
      if (n) n.tombstone = true;
    }
  }

  // Insert node into the backbone right after its origin, then skip past any
  // existing right-siblings that should sort before it (RGA integration rule)
  // — this is what makes concurrent inserts converge regardless of arrival order.
  integrate(node) {
    let pos = node.leftOrigin === null
      ? 0
      : this.nodes.findIndex(n => n.id.site===node.leftOrigin.site && n.id.counter===node.leftOrigin.counter) + 1;
    while (pos < this.nodes.length && this.compareId(this.nodes[pos].id, node.id) < 0) pos++;
    this.nodes.splice(pos, 0, node);
  }

  visibleNodeAt(idx) {
    let seen = -1;
    for (const n of this.nodes) { if (!n.tombstone) { seen++; if (seen === idx) return n; } }
    return null;
  }

  text() {
    return this.nodes.filter(n => !n.tombstone).map(n => n.char).join('');
  }
}
