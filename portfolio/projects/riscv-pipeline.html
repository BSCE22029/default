Hazard Detection + Forwarding Unit (Verilog)
// hazard_unit — stalls on a load-use hazard (real RTL, riscv_pipeline.v)
module hazard_unit (
    input        idex_mem_read,
    input [4:0]  idex_rd,
    input [4:0]  ifid_rs1, ifid_rs2,
    output       stall
);
    assign stall = idex_mem_read &&
                   ((idex_rd == ifid_rs1) || (idex_rd == ifid_rs2)) &&
                   (idex_rd != 5'd0);
endmodule

// forward_unit — EX/MEM has priority over MEM/WB (more recent value)
module forward_unit (
    input [4:0] idex_rs1, idex_rs2,
    input [4:0] exmem_rd, memwb_rd,
    input       exmem_reg_write, memwb_reg_write,
    output reg [1:0] forward_a, forward_b
);
    always @(*) begin
        forward_a = 2'd0;
        if (exmem_reg_write && exmem_rd != 5'd0 && exmem_rd == idex_rs1)
            forward_a = 2'd1;      // forward from EX/MEM
        else if (memwb_reg_write && memwb_rd != 5'd0 && memwb_rd == idex_rs1)
            forward_a = 2'd2;      // forward from MEM/WB
        // forward_b follows the same priority for rs2 (omitted here for brevity)
    end
endmodule
